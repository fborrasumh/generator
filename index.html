// ==================== IndexedDB para Fuentes ====================

let db;

const DB_NAME = "DoctoraliaDB";

const STORE_NAME = "fuentes";


function initDB() {

  const request = indexedDB.open(DB_NAME, 1);

  request.onupgradeneeded = e => {

    db = e.target.result;

    db.createObjectStore(STORE_NAME, { keyPath: "name" });

  };

  request.onsuccess = e => {

    db = e.target.result;

    cargarFuentesGuardadas();

  };

}


async function guardarFuente(file) {

  const text = await file.text();

  const tx = db.transaction(STORE_NAME, "readwrite");

  tx.objectStore(STORE_NAME).put({ 

    name: file.name, 

    content: text, 

    size: Math.round(text.length / 1024) 

  });

  tx.oncomplete = cargarFuentesGuardadas;

}


async function cargarFuentesGuardadas() {

  const tx = db.transaction(STORE_NAME, "readonly");

  const all = await new Promise(r => {

    const req = tx.objectStore(STORE_NAME).getAll();

    req.onsuccess = () => r(req.result);

  });


  let html = "", totalKB = 0;

  all.forEach(f => {

    totalKB += f.size;

    html += `<div class="fuente-item">

      <span>📄 ${f.name} (${f.size} KB)</span>

      <button onclick="eliminarFuente('${f.name}')">Eliminar</button>

    </div>`;

  });


  document.getElementById('listaFuentes').innerHTML = html || "<i>No hay fuentes cargadas</i>";

  document.getElementById('contador').textContent = all.length;

  document.getElementById('tamano').textContent = totalKB;

}


async function eliminarFuente(name) {

  const tx = db.transaction(STORE_NAME, "readwrite");

  tx.objectStore(STORE_NAME).delete(name);

  tx.oncomplete = cargarFuentesGuardadas;

}


function borrarTodasFuentes() {

  if (confirm("¿Eliminar todas las fuentes?")) {

    const tx = db.transaction(STORE_NAME, "readwrite");

    tx.objectStore(STORE_NAME).clear();

    tx.oncomplete = cargarFuentesGuardadas;

  }

}


async function cargarArchivos() {

  const files = document.getElementById('fuentesInput').files;

  for (let file of files) await guardarFuente(file);

  document.getElementById('fuentesInput').value = "";

}


// ==================== API Key ====================

function guardarKey() {

  localStorage.setItem('openai_key', document.getElementById('apiKey').value.trim());

  alert("API Key guardada");

}


// ==================== Llamada a Agente (Streaming) ====================

async function callAgent(systemPrompt, userPrompt, divId, agentName) {

  const apiKey = document.getElementById('apiKey').value.trim() || localStorage.getItem('openai_key');

  if (!apiKey) {

    document.getElementById(divId).innerHTML = "❌ API Key requerida";

    return "";

  }


  // Obtener fuentes

  const tx = db.transaction(STORE_NAME, "readonly");

  const all = await new Promise(r => {

    const req = tx.objectStore(STORE_NAME).getAll();

    req.onsuccess = () => r(req.result);

  });

  const fuentesTexto = all.map(f => `--- FUENTE: ${f.name} ---\n${f.content}`).join("\n\n");


  const div = document.getElementById(divId);

  div.innerHTML = `<strong>${agentName}</strong><div class="streaming">Procesando...</div>`;

  const streamingDiv = div.querySelector('.streaming');


  const fullUserPrompt = `Fuentes proporcionadas:\n${fuentesTexto}\n\n${userPrompt}`;


  const res = await fetch("https://api.openai.com/v1/chat/completions", {

    method: "POST",

    headers: {

      "Content-Type": "application/json",

      "Authorization": `Bearer ${apiKey}`

    },

    body: JSON.stringify({

      model: "gpt-4o-mini",

      temperature: 0.65,

      stream: true,

      messages: [

        { role: "system", content: systemPrompt },

        { role: "user", content: fullUserPrompt }

      ]

    })

  });


  const reader = res.body.pipeThrough(new TextDecoderStream()).getReader();

  let content = "";

  while (true) {

    const {value, done} = await reader.read();

    if (done) break;

    content += value;

    streamingDiv.innerHTML = marked.parse(content);

  }

  return content;

}


// ==================== PIPELINE MULTI-AGENTE COMPLETO ====================

async function iniciarPipeline() {

  const numTemas = parseInt(document.getElementById('numTemas').value);

  const asignatura = document.getElementById('asignatura').value || "Tema General";

  const numApartados = parseInt(document.getElementById('numApartados').value);

  const numPreguntas = parseInt(document.getElementById('numPreguntas').value);


  document.getElementById('output').innerHTML = "<h2>Iniciando Pipeline Multi-Agente completo...</h2>";

  window.notebooks = [];


  for (let t = 1; t <= numTemas; t++) {

    let temaHTML = `<div class="fase"><h2>Tema ${t} — ${asignatura}</h2>`;


    // 1. Planner Agent

    const plannerId = `planner-${t}`;

    temaHTML += `<div class="agente" id="${plannerId}"></div>`;

    document.getElementById('output').innerHTML += temaHTML;


    const estructura = await callAgent(

      "Eres Planner Agent. Experto en diseño pedagógico universitario.",

      `Crea una estructura excelente de exactamente ${numApartados} apartados progresivos para un tema de 1 hora sobre ${asignatura}. Devuelve solo los títulos numerados.`,

      plannerId, "📋 Planner Agent"

    );


    // 2. Map Agent

    const mapId = `map-${t}`;

    temaHTML += `<div class="agente" id="${mapId}"></div>`;

    await callAgent(

      "Eres Map Agent. Experto en visualización de conocimiento.",

      `Genera un mapa conceptual completo en formato Mermaid basado en la estructura anterior.`,

      mapId, "🗺️ Map Agent"

    );


    let notebook = {

      cells: [],

      metadata: { kernelspec: { display_name: "Python 3", language: "python" } },

      nbformat: 4,

      nbformat_minor: 5

    };


    notebook.cells.push({ cell_type: "markdown", source: [`# ${asignatura} - Tema ${t}\n\n**Duración:** 60 minutos`] });


    // Apartados

    for (let a = 1; a <= numApartados; a++) {

      const apId = `ap-${t}-${a}`;

      temaHTML += `<div class="agente" id="${apId}"><h4>Apartado ${a}</h4></div>`;


      // Content Writer

      await callAgent(

        "Eres Content Writer Agent. Profesor universitario claro, riguroso y pedagógico.",

        `Escribe el contenido educativo completo del Apartado ${a}. Usa explicaciones claras, ejemplos y LaTeX.`,

        apId, "✍️ Content Writer Agent"

      );


      // Code & Formula Agent

      await callAgent(

        "Eres Code & Formula Agent. Experto en Python y matemáticas.",

        `Proporciona código Python ejecutable y todas las fórmulas necesarias en LaTeX para este apartado.`,

        apId, "💻 Code & Formula Agent"

      );


      // Exercises Agent

      await callAgent(

        "Eres Exercises Agent. Crea ejercicios bien graduados con resolución detallada.",

        `Crea 3 ejercicios resueltos paso a paso (básico, intermedio, avanzado).`,

        apId, "📝 Exercises Agent"

      );


      // Question Agent

      await callAgent(

        "Eres Question Agent. Especialista en evaluación de calidad.",

        `Crea ${numPreguntas} preguntas tipo test con 4 opciones y explicación detallada de la respuesta correcta.`,

        apId, "❓ Question Agent"

      );


      // Añadir al notebook

      notebook.cells.push({ cell_type: "markdown", source: [`## Apartado ${a}`] });

    }


    // Reviewer Final

    const reviewId = `review-${t}`;

    temaHTML += `<div class="agente" id="${reviewId}"></div>`;

    await callAgent(

      "Eres Reviewer & Polisher Agent. Editor académico exigente.",

      `Revisa todo el tema, mejora coherencia, corrige errores y optimiza el formato para Jupyter.`,

      reviewId, "🔍 Reviewer & Polisher Agent"

    );


    temaHTML += `</div><button onclick="descargarTema(${t-1})">⬇️ Descargar Tema ${t} (.ipynb)</button>`;

    document.getElementById('output').innerHTML += temaHTML;


    window.notebooks.push({ tema: t, notebook: notebook });

  }


  document.getElementById('output').innerHTML += `<br><button onclick="generarZIP()" style="font-size:18px;">📦 Descargar TODOS en ZIP</button>`;

}


// ==================== DESCARGAS ====================

function descargarTema(index) {

  const nb = window.notebooks[index];

  const jsonStr = JSON.stringify(nb.notebook, null, 2);

  const blob = new Blob([jsonStr], { type: "application/json" });

  const url = URL.createObjectURL(blob);

  const a = document.createElement('a');

  a.href = url;

  a.download = `Tema_${nb.tema}.ipynb`;

  a.click();

}


async function generarZIP() {

  const zip = new JSZip();

  window.notebooks.forEach((nb, i) => {

    const jsonStr = JSON.stringify(nb.notebook, null, 2);

    zip.file(`Tema_${nb.tema}.ipynb`, jsonStr);

  });


  const blob = await zip.generateAsync({type: "blob"});

  const url = URL.createObjectURL(blob);

  const a = document.createElement('a');

  a.href = url;

  a.download = `Cuadernos_${document.getElementById('asignatura').value || 'Curso'}.zip`;

  a.click();

}
